def get_met_data(endpoint: str, parameters: dict, Frost_Met_ID: str) -> pd.DataFrame:
    """This function retrives the average air temprature data
    from the frost.met.no API using the provided
    endpoint, parameters, and Frost Met ID. 
    It makes an HTTP GET request to the API using the requests library.
    Stores the data in a json file and transforms it into
    a pandas DataFrame using the pandas library and a loop.
    It then returns a pandas DataFrame containing the retrieved data.
    """
    #Make the API request to retrieve temperature data.
    request = requests.get(endpoint, params=parameters, auth=(Frost_Met_ID, ''))

    #Check the status code of the response and handle errors if any.
    json=request.json()

    #Check if the request was successful and retrieve the data if it wa.
    if request.status_code == 200:
        data = json['data']
        print('Data retrieved from frost.met.no!')
    else:
        print('Error! Returned status code %s' % request.status_code)
        print('Message: %s' % json['error']['message'])
        print('Reason: %s' % json['error']['reason'])

    #Transform the data into a pandas DataFrame for analysis and visualization.
    df = pd.DataFrame()
    for i in range(len(data)):
        row = pd.DataFrame(data[i]['observations'])
        row['referenceTime'] = data[i]['referenceTime']
        row['sourceId'] = data[i]['sourceId']
        df = pd.concat([df, row])

    #Set the index of the DataFrame to the right index for easier analysis.
    df = df.reset_index()
    #Convert the 'referenceTime' column to a datetime object for easier analysis and visualization.
    df['referenceTime'] = pd.to_datetime(df['referenceTime'])
    return df

def number_of_heatwaves(
    endpoint: str,
    parameters: dict,
    ID: str,
    year: int, 
    station_id: str,
    station_name: str = "Ås" 
) -> dict:
    """This function retrives the maximum air temprature data
    from any station using the provided
    endpoint, parameters, and your ID for the station. 
    The function writes a summary to a yml file containing the dates of the heatwaves, station ID, station name and year. 
    Returns the summary as a dictionary
    """
    #Update the parameters to get the correct station and timeperiod
    para = parameters.copy()
    para['sources'] = station_id
    para['referencetime'] = f"{year}-01-01/{year}-12-31"

    #Retrieve data by using an excisting function in this code
    df = get_met_data(endpoint, para, ID)

    if df.empty:
        print("No data to analyse")
        return{}

    #Filtering out only neccesary max-temperatures
    if 'elementId' in df.columns:
        df_temp = df[df['elementId'] == 'max(air_temperature P1D)'].copy()
    else: 
        df_temp = df.copy()

    #Make sure the dates are sorted correctly
    df_temp['referenceTime'] = pd.to_datetime(df_temp['referenceTime'])
    df_temp = df_temp.sort_values('referenceTime').reset_index(drop = True)

    #Convert 'value' to a numerical value
    df_temp['value'] = pd.to_numeric(df_temp['value'], errors = 'coerce')

    #Mark days with temperature >= 27 degrees celsius
    df_temp['is_hot'] = df_temp['value'] >= 27

    #Usa a cumulative sum to group consecutive hot days
    df_temp['streak_group'] = (df_temp['is_hot'] != df_temp['is_hot'].shift()).cumsum()

    heatwaves = []

    #loop through the groupss tp find the groups that lasts for 5 days or more
    for _, group in df_temp.groupby('streak_group'):
        if group['is_hot'].iloc[0] == True and len(group) >= 5:
            start_date = group['referenceTime'].min().strftime('%Y-%m-%d')
            end_date = group['referenceTime'].max().strftime('%Y-%m-%d')
            heatwaves.append(f"{start_date} - {end_date}")

    #Build the structure for the yml-file
    yaml_data = {
        'dates': heatwaves,
        'station_id': station_id,
        'station_name': station_name,
        'year': year
    }

    #Write the summary to the yml-file with the correct format
    filename = f"heatwaves_summary_{station_id}_{year}.yml"
    with open(filename, 'w', encoding = 'utf-8') as f:
        yaml.dump(yaml_data, f, allow_unicode = True, sort_keys = False)

    print(f"Summary saved to {filename}")
    return yaml_data

#Defining a function that calculate a moving avrage fore the values of a pandas DataFrame
def moving_average(df: pd.DataFrame, window_size: int) -> pd.DataFrame:
    """
    This function calculates the moving average of 
    the data using a specified window size. 
    It takes a pandas DataFrame containing the data 
    and an integer representing the window size as input. 
    It uses loops and boolean conditions to calculate the 
    moving average for each data point in the DataFrame.
    It returns a pandas DataFrame containing
    the moving average of the data.
    """

    #Creat an empty list to contain the moving avrages
    moving_avg = []
    #start a loop to iterate through each index of the DataFrame
    for i in range(len(df)): 

        #Check if the current index is less than or equal to the window/2
        if i <= window_size/2:
            #Create a window of values from index 0 to the current index
            window = df['value'][0:i+math.floor(window_size/2)]
            #Calculate the mean of the values in the window
            avrage = window.mean()
            #append the calculated moving average to the list
            moving_avg.append(avrage)

        #Take the current index since it is greater than the window/2
        else:
            #Create a window of values from the current index minus the window size to the current index
            window = df['value'][i - math.floor(window_size/2):i + math.floor(window_size/2)]
            #Calculate the mean of the values in the window
            avrage = window.mean()
            #append the calculated moving average to the list
            moving_avg.append(avrage)
    #Create a new DataFrame to store the moving averages and the corresponding reference times
    moving_avg_df = pd.DataFrame(moving_avg, columns=['moving_average'])
    moving_avg_df['referenceTime'] = df['referenceTime']

    #Return the DataFrame for the moving avrages
    return moving_avg_df
